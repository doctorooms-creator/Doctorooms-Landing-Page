    47→/**
    48→ * AdminOverlay — in-page team panel (no new route, per the single-route rule).
    49→ *
    50→ * Opened by:
    51→ *   • `Shift + A` (registered by `BackToTop` so all global shortcuts live in one place)
    52→ *   • the "Team admin" entry added to the keyboard-shortcuts help dialog
    53→ *   • a `#admin` URL hash deep-link (so the team can bookmark/share)
    54→ *
    55→ * Capabilities:
    56→ *   • Search across name/email/org/orgType
    57→ *   • Filter by status (new / contacted / scheduled / archived)
    58→ *   • Sort by createdAt asc/desc (toggle)
    59→ *   • Inline note editing (PATCH { note }) — team-side context per lead
    60→ *   • Status changes inline (PATCH { status }) — triage workflow
    61→ *   • Batch select + delete (DELETE ?ids=...) for cleanup
    62→ *   • Export filtered rows to CSV
    63→ *   • KPI counts at a glance — click a KPI to filter by that status
    64→ *
    65→ * Reduced-motion safe (CSS only). No auth in this demo sandbox; the
    66→ * comment in the header makes that clear to the team.
    67→ */

    97→function toneClasses(tone: string) {
    98→  switch (tone) {
    99→    case "brand":
   100→      return "border-brand/30 bg-brand-soft/50 text-brand";
   101→    case "amber":
   102→      return "border-amber-500/30 bg-amber-500/10 text-amber-300";
   103→    case "growth":
   104→      return "border-growth/30 bg-growth/10 text-growth";
   105→    default:
   106→      return "border-border/60 bg-muted/40 text-muted-foreground";
   107→  }
   108→}

   110→function statusMeta(v: string) {
   111→  return STATUSES.find((s) => s.value === v) ?? STATUSES[0];

   114→function timeAgo(iso: string) {
   115→  const then = new Date(iso).getTime();
   116→  const diff = Math.max(0, Date.now() - then);
   117→  const min = Math.floor(diff / 60000);
   118→  if (min < 1) return "just now";
   119→  if (min < 60) return `${min}m ago`;
   120→  const hr = Math.floor(min / 60);
   121→  if (hr < 24) return `${hr}h ago`;
   122→  const day = Math.floor(hr / 24);
   123→  if (day < 30) return `${day}d ago`;
   124→  const mo = Math.floor(day / 30);
   125→  return `${mo}mo ago`;
   126→}

   128→function escapeCsv(v: string) {
   129→  if (/[",\n]/.test(v)) return `"${v.replace(/"/g, '""')}"`;
   130→  return v;
   131→}

   133→export function AdminOverlay({

   142→  const [error, setError] = useState<string | null>(null);

   144→  const [filter, setFilter] = useState<string>("all");

   146→  const [updatingId, setUpdatingId] = useState<string | null>(null);
   147→  const [lastFetch, setLastFetch] = useState<number | null>(null);
   148→  const [selected, setSelected] = useState<Set<string>>(new Set());
   149→  const [editingNoteId, setEditingNoteId] = useState<string | null>(null);
   150→  const [noteDraft, setNoteDraft] = useState("");
   151→  const [noteSaving, setNoteSaving] = useState(false);
   152→  const [bulkDeleting, setBulkDeleting] = useState(false);
   153→
   154→  const load = useCallback(async () => {
   155→    setLoading(true);
   156→    setError(null);
   157→    try {
   158→      const res = await fetch("/api/demo", { cache: "no-store" });
   159→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   160→      const json = await res.json();
   161→      setRows(Array.isArray(json.rows) ? (json.rows as Row[]) : []);
   162→      setLastFetch(Date.now());
   163→    } catch (e) {
   164→      setError(e instanceof Error ? e.message : "Failed to load");
   165→    } finally {
   166→      setLoading(false);
   167→    }
   168→  }, []);
   169→
   170→  // Open via #admin hash deep-link.
   171→  useEffect(() => {
   172→    if (typeof window === "undefined") return;
   173→    const checkHash = () => {
   174→      if (window.location.hash === "#admin") {
   175→        track("admin_panel_open", { source: "url_hash" });
   176→        onOpenChange(true);
   177→        // Clean up the hash so subsequent Esc + re-open doesn't auto-trigger.
   178→        try {
   179→          history.replaceState(null, "", window.location.pathname);
   180→        } catch {
   181→          // ignore
   182→        }
   183→      }
   184→    };
   185→    checkHash();
   186→    window.addEventListener("hashchange", checkHash);
   187→    return () => window.removeEventListener("hashchange", checkHash);
   188→  }, [onOpenChange]);
   189→
   190→  useEffect(() => {
   191→    if (open && rows.length === 0 && !loading && !error) {
   192→      void load();
   193→    }
   194→  }, [open, load, rows.length, loading, error]);
   195→
   196→  const filtered = useMemo(() => {
   197→    const q = query.trim().toLowerCase();
   198→    const list = rows.filter((r) => {
   199→      if (filter !== "all" && r.status !== filter) return false;
   200→      if (!q) return true;
   201→      return (
   202→        r.name.toLowerCase().includes(q) ||
   203→        r.email.toLowerCase().includes(q) ||
   204→        r.org.toLowerCase().includes(q) ||
   205→        (r.orgType ?? "").toLowerCase().includes(q)
   206→      );
   207→    });
   208→    list.sort((a, b) => {
   209→      const at = new Date(a.createdAt).getTime();
   210→      const bt = new Date(b.createdAt).getTime();
   211→      return sortDir === "asc" ? at - bt : bt - at;
   212→    });
   213→    return list;
   214→  }, [rows, filter, query, sortDir]);
   215→
   216→  const counts = useMemo(() => {
   217→    const c: Record<string, number> = {
   218→      new: 0,
   219→      contacted: 0,
   220→      scheduled: 0,
   221→      archived: 0,
   222→    };
   223→    for (const r of rows) {
   224→      if (c[r.status] != null) c[r.status] += 1;
   225→    }
   226→    return c;
   227→  }, [rows]);
   228→
   229→  const allFilteredSelected =
   230→    filtered.length > 0 && filtered.every((r) => selected.has(r.id));
   231→
   232→  function toggleSelect(id: string) {
   233→    setSelected((prev) => {
   234→      const next = new Set(prev);
   235→      if (next.has(id)) next.delete(id);
   236→      else next.add(id);
   237→      return next;
   238→    });
   239→  }
   240→
   241→  function toggleSelectAll() {
   242→    if (allFilteredSelected) {
   243→      setSelected(new Set());
   244→    } else {
   245→      setSelected(new Set(filtered.map((r) => r.id)));
   246→    }
   247→  }
   248→
   249→  async function changeStatus(r: Row, next: string) {
   250→    setUpdatingId(r.id);
   251→    try {
   252→      const res = await fetch(`/api/demo?id=${encodeURIComponent(r.id)}`, {
   253→        method: "PATCH",
   254→        headers: { "Content-Type": "application/json" },
   255→        body: JSON.stringify({ status: next }),
   256→      });
   257→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   258→      const json = await res.json();
   259→      const updated = json.row as Row | undefined;
   260→      setRows((prev) =>
   261→        prev.map((row) =>
   262→          row.id === r.id ? { ...row, status: updated?.status ?? next } : row
   263→        )
   264→      );
   265→      track("admin_status_change", {
   266→        id: r.id,
   267→        from: r.status,
   268→        to: next,
   269→        org: r.org,
   270→      });
   271→    } catch (e) {
   272→      setError(e instanceof Error ? e.message : "Failed to update");
   273→    } finally {
   274→      setUpdatingId(null);
   275→    }
   276→  }
   277→
   278→  async function saveNote(r: Row, override?: string) {
   279→    setNoteSaving(true);
   280→    // Use override if provided (Clear button) so we don't race against
   281→    // async setState; otherwise use the live textarea draft.
   282→    const value = override !== undefined ? override : noteDraft;
   283→    try {
   284→      const res = await fetch(`/api/demo?id=${encodeURIComponent(r.id)}`, {
   285→        method: "PATCH",
   286→        headers: { "Content-Type": "application/json" },
   287→        body: JSON.stringify({ note: value }),
   288→      });
   289→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   290→      const json = await res.json();
   291→      const updated = json.row as Row | undefined;
   292→      setRows((prev) =>
   293→        prev.map((row) =>
   294→          row.id === r.id
   295→            ? { ...row, note: updated?.note ?? value }
   296→            : row
   297→        )
   298→      );
   299→      track("admin_note_update", {
   300→        id: r.id,
   301→        org: r.org,
   302→        cleared: !value.trim(),
   303→      });
   304→      setEditingNoteId(null);
   305→    } catch (e) {
   306→      setError(e instanceof Error ? e.message : "Failed to save note");
   307→    } finally {
   308→      setNoteSaving(false);
   309→    }
   310→  }
   311→
   312→  async function bulkDelete() {
   313→    if (selected.size === 0) return;
   314→    const ids = Array.from(selected);
   315→    if (
   316→      !window.confirm(
   317→        `Delete ${ids.length} selected demo request${ids.length === 1 ? "" : "s"}? This cannot be undone.`
   318→      )
   319→    ) {
   320→      return;
   321→    }
   322→    setBulkDeleting(true);
   323→    try {
   324→      const res = await fetch(
   325→        `/api/demo?ids=${ids.map(encodeURIComponent).join(",")}`,
   326→        { method: "DELETE" }
   327→      );
   328→      if (!res.ok) throw new Error(`HTTP ${res.status}`);
   329→      const json = await res.json();
   330→      const deleted = (json.deleted as number | undefined) ?? 0;
   331→      setRows((prev) => prev.filter((r) => !selected.has(r.id)));
   332→      track("admin_bulk_delete", { count: deleted });
   333→      setSelected(new Set());
   334→    } catch (e) {
   335→      setError(e instanceof Error ? e.message : "Failed to delete");
   336→    } finally {
   337→      setBulkDeleting(false);
   338→    }
   339→  }
   340→
   341→  async function archiveSelected() {
   342→    if (selected.size === 0) return;
   343→    setBulkDeleting(true);
   344→    try {
   345→      // Sequential PATCHes — small batch sizes in practice.
   346→      const ids = Array.from(selected);
   347→      await Promise.all(
   348→        ids.map(async (id) => {
   349→          const r = rows.find((row) => row.id === id);
   350→          if (!r || r.status === "archived") return;
   351→          const res = await fetch(`/api/demo?id=${encodeURIComponent(id)}`, {
   352→            method: "PATCH",
   353→            headers: { "Content-Type": "application/json" },
   354→            body: JSON.stringify({ status: "archived" }),
   355→          });
   356→          if (res.ok) {
   357→            track("admin_status_change", {
   358→              id,
   359→              from: r.status,
   360→              to: "archived",
   361→              org: r.org,
   362→            });
   363→          }
   364→        })
   365→      );
   366→      // Refresh from server to get authoritative state.
   367→      await load();
   368→      setSelected(new Set());
   369→    } catch (e) {
   370→      setError(e instanceof Error ? e.message : "Failed to archive");
   371→    } finally {
   372→      setBulkDeleting(false);
   373→    }
   374→  }

   376→  function exportCsv() {
   377→    const header = [
   378→      "name",
   379→      "email",
   380→      "phone",
   381→      "org",
   382→      "orgType",
   383→      "size",
   384→      "status",
   385→      "createdAt",
   386→      "note",
   387→    ];
   388→    const lines = filtered.map((r) =>
   389→      [
   390→        r.name,
   391→        r.email,
   392→        r.phone ?? "",
   393→        r.org,
   394→        r.orgType ?? "",
   395→        r.size ?? "",
   396→        r.status,
   397→        new Date(r.createdAt).toISOString(),
   398→        r.note ?? "",
   399→      ]
   400→        .map(escapeCsv)
   401→        .join(",")
   402→    );
   403→    const csv = [header.join(","), ...lines].join("\n");
   404→    const blob = new Blob([csv], { type: "text/csv;charset=utf-8" });
   405→    const url = URL.createObjectURL(blob);
   406→    const a = document.createElement("a");
   407→    a.href = url;
   408→    a.download = `doctorooms-demo-requests-${new Date()
   409→      .toISOString()
   410→      .slice(0, 10)}.csv`;
   411→    document.body.appendChild(a);
   412→    a.click();
   413→    a.remove();
   414→    URL.revokeObjectURL(url);
   415→  }

   423→        <div className="flex max-h-[92vh] flex-col">
   424→          {/* Header */}
   425→          <DialogHeader className="border-b border-border/60 bg-gradient-to-br from-brand-soft/40 to-transparent px-5 py-4 sm:px-6">
   426→            <DialogTitle className="flex flex-wrap items-center gap-2 text-xl">
   427→              <span className="flex h-8 w-8 items-center justify-center rounded-lg bg-brand-soft text-brand">
   428→                <ShieldCheck className="h-4 w-4" />
   429→              </span>
   430→              Team admin
   431→              <Badge className="ml-1 border-border/60 bg-muted/40 text-[10px] font-normal text-muted-foreground">
   432→                demo-requests
   433→              </Badge>
   434→            </DialogTitle>
   435→            <DialogDescription id="admin-overlay-desc" className="text-xs">
   436→              Inbound &quot;Book a Private Demo&quot; submissions, persisted to the
   437→              landing-page SQLite database. No auth in this sandbox view —

   520→            </Button>
   521→            <Button
   522→              type="button"
   523→              variant="outline"
   524→              size="sm"
   525→              onClick={() => void load()}
   526→              disabled={loading}
   527→              className="h-9"

   530→              Refresh

   532→            <Button
   533→              type="button"

   577→                variant="outline"
   578→                size="sm"
