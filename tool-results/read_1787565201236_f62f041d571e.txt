    51→/**
    52→ * AdminOverlay — in-page team panel (no new route, per the single-route rule).
    53→ *
    54→ * Opened by:
    55→ *   • `Shift + A` (registered by `BackToTop` so all global shortcuts live in one place)
    56→ *   • the "Team admin" entry added to the keyboard-shortcuts help dialog
    57→ *   • a `#admin` URL hash deep-link (so the team can bookmark/share)
    58→ *
    59→ * Capabilities:
    60→ *   • Search across name/email/org/orgType
    61→ *   • Filter by status (new / contacted / scheduled / archived)
    62→ *   • Sort by Date / Name / Org, asc or desc
    63→ *   • Inline note editing (PATCH { note }) — team-side context per lead
    64→ *   • Status changes inline (PATCH { status }) — triage workflow
    65→ *   • Batch select + delete (DELETE ?ids=...) for cleanup
    66→ *   • Per-row Copy email + Open mailto (pre-filled compose) + bulk Copy selected emails
    67→ *   • Export filtered rows to CSV (flat) or by-status (grouped w/ status column)
    68→ *   • KPI counts at a glance — click a KPI to filter by that status
    69→ *
    70→ * Reduced-motion safe (CSS only). No auth in this demo sandbox; the
    71→ * comment in the header makes that clear to the team.
    72→ */

   119→function timeAgo(iso: string) {
   120→  const then = new Date(iso).getTime();
   121→  const diff = Math.max(0, Date.now() - then);
   122→  const min = Math.floor(diff / 60000);
   123→  if (min < 1) return "just now";
   124→  if (min < 60) return `${min}m ago`;
   125→  const hr = Math.floor(min / 60);
   126→  if (hr < 24) return `${hr}h ago`;
   127→  const day = Math.floor(hr / 24);
   128→  if (day < 30) return `${day}d ago`;
   129→  const mo = Math.floor(day / 30);
   130→  return `${mo}mo ago`;
   131→}

   133→function escapeCsv(v: string) {
   134→  if (/[",\n]/.test(v)) return `"${v.replace(/"/g, '""')}"`;
   135→  return v;
   136→}

   356→  async function archiveSelected() {
   357→    if (selected.size === 0) return;

   359→    try {
   360→      // Sequential PATCHes — small batch sizes in practice.
   361→      const ids = Array.from(selected);
   362→      await Promise.all(
   363→        ids.map(async (id) => {
   364→          const r = rows.find((row) => row.id === id);

   367→            method: "PATCH",
   368→            headers: { "Content-Type": "application/json" },
   369→            body: JSON.stringify({ status: "archived" }),
   370→          });
   371→          if (res.ok) {
   372→            track("admin_status_change", {
   373→              id,
   374→              from: r.status,
   375→              to: "archived",
   376→              org: r.org,
   377→            });
   378→          }
   379→        })
   380→      );
   381→      // Refresh from server to get authoritative state.
   382→      await load();
   383→      setSelected(new Set());
   384→    } catch (e) {
   385→      setError(e instanceof Error ? e.message : "Failed to archive");
   386→    } finally {
   387→      setBulkDeleting(false);

   436→  function exportCsvByStatus() {

   439→    const header = ["status", "name", "email", "phone", "org", "orgType", "size", "createdAt", "note"];

   444→    });

   449→        r.email,

   458→        .join(",")
   459→    );
   460→    const csv = [header.join(","), ...lines].join("\n");
   461→    const blob = new Blob([csv], { type: "text/csv;charset=utf-8" });
   462→    const url = URL.createObjectURL(blob);

   490→  async function copyEmail(r: Row) {
   491→    try {
   492→      await navigator.clipboard.writeText(r.email);
   493→    } catch {
   494→      const ta = document.createElement("textarea");
   495→      ta.value = r.email;
   496→      ta.style.position = "fixed";
   497→      ta.style.opacity = "0";
   498→      document.body.appendChild(ta);
   499→      ta.select();
   500→      try {
   501→        document.execCommand("copy");
   502→      } catch {
   503→        // give up
   504→      }
   505→      ta.remove();
   506→    }
   507→    setCopiedId(r.id);
   508→    track("admin_email_copy", { id: r.id });
   509→    setTimeout(() => setCopiedId((cur) => (cur === r.id ? null : cur)), 1600);
   510→  }
